A pico2 uf2 file is found at
https://github.com/develone/hsdaoh-rp2350/blob/test-dev/apps-uf2-files/counter/counter.uf2

To build hsdaoh-rp2350:  After creating a fork of the repo and the test-dev branch.
devel@pi5-80:~/hsdaoh-rp2350/build $ git clone -b test-dev https://github.com/develone/hsdaoh-rp2350.git

devel@pi5-80:~/hsdaoh-rp2350/build $ mkdir hsdaoh-rp2350/build

devel@pi5-80:~/hsdaoh-rp2350/build $ cd hsdaoh-rp2350/build
 
devel@pi5-80:~/hsdaoh-rp2350/build $ cmake -DPICO_PLATFORM=rp2350 -DPICO_BOARD=pico2 ../

devel@pi5-80:~/hsdaoh-rp2350/build $ make -j 8

devel@pi5-80:~/hsdaoh-rp2350/build $ ls apps/counter/
CMakeFiles           counter.bin  counter.elf      counter.hex    counter.uf2
cmake_install.cmake  counter.dis  counter.elf.map  counter.pio.h  Makefile

devel@pi5-80:~/hsdaoh-rp2350/build $ cp apps/counter/counter.uf2 /media/devel/RP2350
